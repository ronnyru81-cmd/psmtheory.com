/* AVA EVENT LOG v1 – live instrumentation of every web / research action.
   Schema (one object per event):
   EVENT_ID, TIMESTAMP, TRIGGER_TYPE (autonomous|manual|passive), QUERY, PROVIDER,
   REQUEST_STARTED, HTTP_STATUS, REQUEST_COMPLETED, RESULT_COUNT, SOURCE_URLS,
   CONTENT_LENGTH, PAGE_FETCHED, MEMORY_WRITE, MEMORY_ID, ERROR
*/
(function () {
  'use strict';
  const KEY = 'ava_public_eventlog_v1';
  const MAX = 800;
  const $ = id => document.getElementById(id);
  const esc = s => String(s == null ? '' : s).replace(/[&<>"']/g, c => ({ '&':'&','<':'<','>':'>','"':'"',"'":''' }[c]));
  let events = [];
  let seq = 0;
  let enabled = true;
  try {
    const raw = localStorage.getItem(KEY);
    if (raw) {
      const j = JSON.parse(raw);
      events = Array.isArray(j.events) ? j.events.slice(0, MAX) : [];
      seq = j.seq || events.length;
      enabled = j.enabled !== false;
    }
  } catch (e) {}

  function uid() {
    seq++;
    const t = Date.now().toString(36);
    return 'ev_' + t + '_' + seq.toString(36);
  }
  function nowIso() { return new Date().toISOString(); }
  function scheduleSave() {
    clearTimeout(scheduleSave._t);
    scheduleSave._t = setTimeout(() => {
      try { localStorage.setItem(KEY, JSON.stringify({ seq, enabled, events: events.slice(0, MAX) })); } catch (e) {}
    }, 1200);
  }

  function push(ev) {
    if (!enabled) return ev;
    events.unshift(ev);
    if (events.length > MAX) events.length = MAX;
    scheduleSave();
    refreshBadge();
    if ($('avaLogModal') && !$('avaLogModal').classList.contains('hidden')) render();
    return ev;
  }

  /** Start a timed event. Returns the event object; call finish() when done. */
  function begin(opts) {
    opts = opts || {};
    const trigger = window.__avaNextTrigger || opts.trigger || 'autonomous';
    const ev = {
      EVENT_ID: uid(),
      TIMESTAMP: nowIso(),
      TRIGGER_TYPE: trigger,   // autonomous | manual | passive
      QUERY: String(opts.query || '').slice(0, 300),
      PROVIDER: opts.provider || '',
      REQUEST_STARTED: nowIso(),
      HTTP_STATUS: null,
      REQUEST_COMPLETED: null,
      RESULT_COUNT: 0,
      SOURCE_URLS: [],
      CONTENT_LENGTH: 0,
      PAGE_FETCHED: false,
      MEMORY_WRITE: false,
      MEMORY_ID: null,
      ERROR: null,
      _meta: { via: opts.via || '', notes: opts.notes || '' }
    };
    push(ev);
    return ev;
  }

  function finish(ev, patch) {
    if (!ev) return;
    Object.assign(ev, patch || {});
    ev.REQUEST_COMPLETED = nowIso();
    if (Array.isArray(ev.SOURCE_URLS)) {
      ev.SOURCE_URLS = ev.SOURCE_URLS.filter(Boolean).map(u => String(u).slice(0, 300)).slice(0, 12);
    }
    scheduleSave();
    refreshBadge();
    if ($('avaLogModal') && !$('avaLogModal').classList.contains('hidden')) render();
  }

  // ---------- public API ----------
  window.avaEventLog = {
    begin, finish,
    log(opts) {                       // fire-and-forget complete event
      const ev = begin(opts);
      finish(ev, opts);
      return ev;
    },
    events: () => events.slice(),
    clear() {
      if (!confirm('Clear all ' + events.length + ' logged events?')) return;
      events = []; scheduleSave(); refreshBadge(); render();
    },
    setEnabled(v) { enabled = !!v; scheduleSave(); refreshBadge(); },
    isEnabled: () => enabled,
    exportJson() {
      const blob = new Blob([JSON.stringify(events, null, 2)], { type: 'application/json' });
      const a = document.createElement('a');
      a.href = URL.createObjectURL(blob);
      a.download = 'ava_eventlog_' + new Date().toISOString().slice(0, 19).replace(/[:T]/g, '-') + '.json';
      a.click(); setTimeout(() => URL.revokeObjectURL(a.href), 2000);
    },
    exportCsv() {
      const cols = ['EVENT_ID','TIMESTAMP','TRIGGER_TYPE','QUERY','PROVIDER','REQUEST_STARTED','HTTP_STATUS','REQUEST_COMPLETED','RESULT_COUNT','SOURCE_URLS','CONTENT_LENGTH','PAGE_FETCHED','MEMORY_WRITE','MEMORY_ID','ERROR'];
      const escCsv = v => {
        if (v == null) return '';
        const s = Array.isArray(v) ? v.join(' | ') : String(v);
        return /[",\n]/.test(s) ? '"' + s.replace(/"/g, '""') + '"' : s;
      };
      const lines = [cols.join(',')].concat(events.map(e => cols.map(c => escCsv(e[c])).join(',')));
      const blob = new Blob([lines.join('\n')], { type: 'text/csv' });
      const a = document.createElement('a');
      a.href = URL.createObjectURL(blob);
      a.download = 'ava_eventlog_' + new Date().toISOString().slice(0, 10) + '.csv';
      a.click(); setTimeout(() => URL.revokeObjectURL(a.href), 2000);
    },
    open() { $('avaLogModal').classList.remove('hidden'); render(); },
    close() { $('avaLogModal').classList.add('hidden'); }
  };

  // ---------- instrument core web paths ----------
  function inferTrigger(manualFlag, opts) {
    if (manualFlag === true || (opts && opts.manual)) return 'manual';
    if (opts && opts.passive) return 'passive';
    return 'autonomous';
  }

  // Wrap researchWeb – the central research entry point
  const origResearchWeb = window.researchWeb;
  if (typeof origResearchWeb === 'function') {
    window.researchWeb = async function (query, opts) {
      opts = opts || {};
      // Prefer explicit one-shot flag, then opts.manual, then autonomous
      const trigger = window.__avaNextTrigger || inferTrigger(opts.manual, opts);
      const providerGuess = (typeof nativeWebKind === 'function' && nativeWebKind()) || (llm && llm.provider) || 'free';
      const ev = begin({
        query: query,
        trigger,
        provider: providerGuess,
        notes: opts.news ? 'news' : (opts.native === false ? 'free-only' : '')
      });
      try {
        const res = await origResearchWeb.apply(this, arguments);
        const sources = (res && res.sources) || [];
        const urls = sources.map(s => s.uri || s.url).filter(Boolean);
        const textLen = (res && res.text) ? res.text.length : 0;
        finish(ev, {
          PROVIDER: (res && res.via) || providerGuess,
          RESULT_COUNT: sources.length || (res && res.text ? 1 : 0),
          SOURCE_URLS: urls,
          CONTENT_LENGTH: textLen,
          ERROR: (res && res.errors && res.errors.length) ? res.errors.slice(0, 3).join('; ') : null,
          HTTP_STATUS: (res && res.insufficient) ? 204 : 200
        });
        return res;
      } catch (e) {
        finish(ev, { ERROR: e.message || String(e), HTTP_STATUS: e.status || null });
        throw e;
      }
    };
  }

  // Wrap fetchViaProxy so page fetches are recorded
  const origFetchViaProxy = window.fetchViaProxy;
  if (typeof origFetchViaProxy === 'function') {
    window.fetchViaProxy = async function (url) {
      const ev = begin({
        query: url,
        trigger: 'passive',
        provider: 'proxy',
        notes: 'page-fetch'
      });
      try {
        const text = await origFetchViaProxy.apply(this, arguments);
        finish(ev, {
          HTTP_STATUS: 200,
          CONTENT_LENGTH: (text || '').length,
          PAGE_FETCHED: true,
          RESULT_COUNT: 1,
          SOURCE_URLS: [url]
        });
        return text;
      } catch (e) {
        finish(ev, { ERROR: e.message || String(e), PAGE_FETCHED: false, HTTP_STATUS: null });
        throw e;
      }
    };
  }

  // Wrap performOpenWebSearch (manual toolbar button)
  const origOpenSearch = window.performOpenWebSearch;
  if (typeof origOpenSearch === 'function') {
    window.performOpenWebSearch = async function () {
      // researchWeb already logs; just mark the next research as manual via a one-shot flag
      window.__avaNextTrigger = 'manual';
      try { return await origOpenSearch.apply(this, arguments); }
      finally { delete window.__avaNextTrigger; }
    };
  }

  // processAutonomousInquiry – curiosity / autonomous
  const origInquiry = window.processAutonomousInquiry;
  if (typeof origInquiry === 'function') {
    window.processAutonomousInquiry = async function (topicQuery, opts) {
      opts = opts || {};
      if (opts.manual) window.__avaNextTrigger = 'manual';
      else window.__avaNextTrigger = 'autonomous';
      try { return await origInquiry.apply(this, arguments); }
      finally { delete window.__avaNextTrigger; }
    };
  }

  // runNewsPulse – passive/autonomous news
  const origNews = window.runNewsPulse;
  if (typeof origNews === 'function') {
    window.runNewsPulse = async function (manual) {
      window.__avaNextTrigger = manual ? 'manual' : 'passive';
      try { return await origNews.apply(this, arguments); }
      finally { delete window.__avaNextTrigger; }
    };
  }

  // begin already reads window.__avaNextTrigger when present (see begin implementation)

  // Memory writes – hook addMemory / state.memories push when present
  const origAddMemory = window.addMemory;
  if (typeof origAddMemory === 'function') {
    window.addMemory = function (text, status, source) {
      const id = 'mem_' + Date.now().toString(36);
      const r = origAddMemory.apply(this, arguments);
      // attach to most recent open event if any
      const last = events[0];
      if (last && !last.MEMORY_WRITE && last.REQUEST_COMPLETED) {
        last.MEMORY_WRITE = true;
        last.MEMORY_ID = id;
        scheduleSave();
      } else {
        // standalone memory event
        push({
          EVENT_ID: uid(), TIMESTAMP: nowIso(), TRIGGER_TYPE: 'passive',
          QUERY: String(text || '').slice(0, 120), PROVIDER: 'memory',
          REQUEST_STARTED: nowIso(), REQUEST_COMPLETED: nowIso(),
          HTTP_STATUS: null, RESULT_COUNT: 1, SOURCE_URLS: source ? [String(source).slice(0, 200)] : [],
          CONTENT_LENGTH: String(text || '').length, PAGE_FETCHED: false,
          MEMORY_WRITE: true, MEMORY_ID: id, ERROR: null
        });
      }
      return r;
    };
  }

  // Also note value-memory / study writes if avaNet.remember is used
  const tryHookNet = () => {
    if (!window.avaNet || window.avaNet.__logged) return;
    const orig = window.avaNet.remember;
    if (typeof orig !== 'function') return;
    window.avaNet.remember = function (text, imp, src, why, url) {
      const r = orig.apply(this, arguments);
      const last = events[0];
      if (last && !last.MEMORY_WRITE) {
        last.MEMORY_WRITE = true;
        last.MEMORY_ID = (r && r.id) ? 'vn_' + r.id : 'vn_' + Date.now().toString(36);
        scheduleSave();
      }
      return r;
    };
    window.avaNet.__logged = true;
  };
  setTimeout(tryHookNet, 2500);
  setTimeout(tryHookNet, 8000);

  // ---------- UI ----------
  function refreshBadge() {
    const b = $('avaLogBtn');
    if (!b) return;
    const n = events.length;
    b.textContent = '📋 Log ' + n + (enabled ? '' : ' (off)');
  }

  function ago(iso) {
    if (!iso) return '–';
    const ms = Date.now() - new Date(iso).getTime();
    if (ms < 60000) return Math.round(ms / 1000) + 's';
    if (ms < 3600000) return Math.round(ms / 60000) + 'm';
    if (ms < 86400000) return Math.round(ms / 3600000) + 'h';
    return Math.round(ms / 86400000) + 'd';
  }

  function triggerColor(t) {
    return t === 'manual' ? 'text-amber-300' : t === 'passive' ? 'text-slate-400' : 'text-emerald-300';
  }

  function render() {
    const body = $('avaLogBody');
    if (!body) return;
    const filter = ($('avaLogFilter') && $('avaLogFilter').value) || 'all';
    let rows = events;
    if (filter === 'manual') rows = events.filter(e => e.TRIGGER_TYPE === 'manual');
    else if (filter === 'autonomous') rows = events.filter(e => e.TRIGGER_TYPE === 'autonomous');
    else if (filter === 'passive') rows = events.filter(e => e.TRIGGER_TYPE === 'passive');
    else if (filter === 'error') rows = events.filter(e => e.ERROR);
    else if (filter === 'memory') rows = events.filter(e => e.MEMORY_WRITE);

    body.innerHTML =
      '
' +
      '' +
      '' +
      '' + rows.length + ' shown / ' + events.length + ' total' +
      '' +
      '' +
      '' +
      '
' +
      '
' +
      'SchemaEVENT_ID · TIMESTAMP · TRIGGER_TYPE · QUERY · PROVIDER · REQUEST_STARTED · HTTP_STATUS · REQUEST_COMPLETED · RESULT_COUNT · SOURCE_URLS · CONTENT_LENGTH · PAGE_FETCHED · MEMORY_WRITE · MEMORY_ID · ERROR' +
      '
' +
      (rows.length ? rows.map(e => {
        const dur = (e.REQUEST_STARTED && e.REQUEST_COMPLETED)
          ? Math.max(0, new Date(e.REQUEST_COMPLETED) - new Date(e.REQUEST_STARTED)) + ' ms' : '–';
        return '
' +
          '
' +
          '' + esc(e.EVENT_ID) + '' +
          '' + esc(e.TRIGGER_TYPE) + '' +
          '' + ago(e.TIMESTAMP) + ' ago' +
          (e.PROVIDER ? '' + esc(e.PROVIDER) + '' : '') +
          (e.HTTP_STATUS != null ? 'HTTP ' + e.HTTP_STATUS + '' : '') +
          '' + dur + '' +
          (e.PAGE_FETCHED ? 'page' : '') +
          (e.MEMORY_WRITE ? 'mem ' + esc(e.MEMORY_ID || '') + '' : '') +
          (e.ERROR ? '⚠ error' : '') +
          '
' +
          '
Q: ' + esc(e.QUERY || '–') + '
' +
          '
results ' + (e.RESULT_COUNT || 0) +
          ' · content ' + (e.CONTENT_LENGTH || 0) + ' B' +
          (e.SOURCE_URLS && e.SOURCE_URLS.length ? ' · ' + e.SOURCE_URLS.slice(0, 3).map(u => '' + esc(u.slice(0, 60)) + (u.length > 60 ? '…' : '') + '').join(' · ') : '') +
          '
' +
          (e.ERROR ? '
' + esc(e.ERROR) + '
' : '') +
          '
';
      }).join('') : '
No events yet. Web searches, news pulses, page fetches and memory writes will appear here.
');
  }
  window.renderLog = render;

  function mount() {
    const pb = $('pauseBtn');
    if (!pb || $('avaLogBtn')) return;
    const b = document.createElement('button');
    b.id = 'avaLogBtn';
    b.type = 'button';
    b.className = 'bg-slate-800 hover:bg-slate-700 border border-slate-600 text-slate-200 text-[10px] px-2 py-0.5 rounded transition font-bold';
    b.onclick = () => window.avaEventLog.open();
    pb.parentNode.insertBefore(b, pb);

    const m = document.createElement('div');
    m.id = 'avaLogModal';
    m.className = 'hidden fixed inset-0 z-50 bg-black/70 flex items-start justify-center p-3 overflow-y-auto';
    m.onclick = e => { if (e.target === m) window.avaEventLog.close(); };
    m.innerHTML =
      '
' +
      '
' +
      '
📋 Event Log – web & research activity
' +
      '
' +
      '
';
    document.body.appendChild(m);
    refreshBadge();
  }

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', mount);
  else mount();
  // re-try badge after other modules mount their buttons
  setTimeout(mount, 1500);
  setTimeout(refreshBadge, 2000);
})();
